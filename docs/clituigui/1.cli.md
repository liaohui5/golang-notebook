## 常见命令应用的2种形式

1. 只有一个 `主命令` + `参数` + `操作目标(无名参数)`

```sh
curl -i -X GET http://localhost:3000/api

# 主命令: curl
# 参数: -i -X
# 无名参数(操作目标): http://localhost:3000/api
```

2. 除 `主命令` 入口以外还有 `子命令及其参数`

```sh
git --help
git add --help
git commit --help
# 主命令: git
# 子命令: add/commit
# 主命令+参数: git --help
# 子命令+参数: add --help / commit --help
```

## 对比参数解析库优劣

### flags

一些简单的参数解析可以用这个标准库, 但是如果要支持类似 `git add xxx` 这样的子命令, 实现比较复杂, 功能较少

- 标准库文档: https://pkg.go.dev/flag

### urfave/cli

- 源码链接: https://github.com/urfave/cli
- 官方文档: https://cli.urfave.org/
- 优势: 声明式风格, 代码直观明了，一个 struct 搞定，代码量少, 少量命令时结构清晰
- 缺点: 多级子命令嵌套时 struct 层级很深, 可读性差

### spf13/cobra

- 源码链接: https://github.com/spf13/cobra
- 官方文档: https://cobra.dev/
- 优势: 天然支持任意深度的子命令嵌套
- 缺点: 学习成本高, 概念很多(Command/Flag/PersistentFlag/Args/Template)
- 额外依赖: [spf13/pflag](https://github.com/spf13/pflag) 它与标准库 [flag](https://pkg.go.dev/flag) 并不兼容

### textfmt

尝试实现一个文本格式化命令, 并且包含 `upper` `lower` `contains` 3个子命令, 并且有 `--debug/-d` 全局选项

- upper: 将输入的文本转为大写 `--json` 是否以json格式输出结果(局部选项)
- lower: 将输入的文本转为小写 `--json` 是否以json格式输出结果(局部选项)
- contains: 判断输入的文本是否包含某个子文本
  - `--ignore/-i` 是否忽略大小写查找
  - `--keyword/-k` 要搜索的关键字

::: code-group

```sh [示例用法]
textfmt upper --debug "Hello World"
# 输出: HELLO WORLD

textfmt upper -d --json "Hello World"
# 输出: {"original":"Hello World", "formatted": "HELLO WORLD", isDebug:true}

textfmt lower "Hello World" --json
# 输出: {"original":"Hello World", "formatted": "Hello World", isDebug:false}

textfmt contains "Hello World" --keyword="hello"
# 输出: false, 默认情况是不忽略大小写

textfmt contains "Hello World" -k="hello" -i
# 输出: true, 忽略大小写, 就能找到关键字 `hello`
```

```go [修改 fanyi.go]
package main

import (
	"context"
	"fmt"
	"log"
	"os"
	"slices"

	"github.com/urfave/cli/v3"
)

// 定义选项 & 默认值
var (
	provider string = "openai" // 可以这样设置默认值(注意使用这个变量设置 cli.StringFlag#Value 字段)
	prompt   string
	isDebug  bool
)

// 定义允许的翻译提供商方便验证传入的参数值是否有误
var allowProviders = []string{
	"baidu",
	"youdao",
	"openai",
	"deepseek",
	"openai_compatible",
}

func main() {
	flags := []cli.Flag{
		// --provider
		&cli.StringFlag{
			Name:        "provider",                 // 选项名称 --provider
			Aliases:     []string{"p"},              // 选项短别名 -p
			Value:       provider,                   // 选项默认值(使用 provider 变量)
			Usage:       "translation api provider", // 使用说明
			Destination: &provider,                  // 绑定的变量地址
		},

		// --prompt
		&cli.StringFlag{
			Name:        "prompt",
			Value:       "Please translate this text, output only the translation result.",
			Usage:       "translation api provider",
			Destination: &prompt,
		},

		// --debug
		&cli.BoolFlag{
			Name:        "debug",
			Aliases:     []string{"d"},
			Value:       false,
			Usage:       "print debug information",
			Destination: &isDebug,
		},
	}

	// 定义命令名称/描述/执行方法
	cmd := &cli.Command{
		Flags: flags,
		Name:  "fanyi",
		Usage: "translate text in command line",
		Action: func(ctx context.Context, cmd *cli.Command) error {
			// 待翻译的文本验证规格
			if cmd.NArg() <= 0 {
				return fmt.Errorf("missing text to translate")
			}
			text := cmd.Args().Get(0)

			// --provider 参数校验
			if !slices.Contains(allowProviders, provider) {
				return fmt.Errorf("invalid translation api provider: %s", provider)
			}

			fmt.Printf("待翻译的文本是:%s \n使用的API是:%s\nprompt内容是:%s\nisDebug:%v\n", text, provider, prompt, isDebug)

			// 所有参数已经收集完成, 后续具体实现就是:
			// 0.安装 viper, 读取配置文件内容
			// 1.从 ~/.config/fanyi/config.toml 中读取 provider 配置
			// 2.判断是否是大模型接口, 如果是, 则需要 prompt 参数
			// 3.按照读取到的 provider 参数, 创建对应的处理方法(请求不同的接口&输出结果)
			// 然而这些并不是学习 cobra/cli 的重点, 感兴趣的话可以自己实现下
			return nil
		},
	}

	if err := cmd.Run(context.Background(), os.Args); err != nil {
		log.Fatal(err)
	}
}
```

```go [fanyi2/cmd/root.go]
	isDebug  bool   = false                                                             // 是否输出调试信息
)

// 允许的接口提供商
var allowProviders = []string{
	"baidu",
	"youdao",
	"openai",
	"deepseek",
	"openai_compatible",

// 定义命令

// init 模块初始化时定义命令的参数
func init() {

	// StringVarP(选项的引用, 选项名, 短选项名, 选项描述)

	rootCmd.Flags().BoolVarP(&isDebug, "debug", "d", isDebug, "print debug infomation")

	}

	// provider: 参数校验
	if !slices.Contains(allowProviders, provider) {
		return fmt.Errorf("[Error]Invalid translation api provider: %s", provider)
	}

	// 具体实现就不实现了, 本质就是根据收集到的参数,
	// 执行请求不同的接口去翻译文本, 这里用输出函数代替
	// 待翻译的文本是: $text
	// 使用的API是: $provider
	// prompt内容是: $prompt
	// 是否输出调试信息: $isDebug
	fmt.Printf("待翻译的文本是:%s \n使用的API是:%s\nprompt内容是:%s\nisDebug:%v\n", text, provider, prompt, isDebug)
```

## urfave/cli 实现 textfmt

::: code-group

```go [textfmt.go]
package main

import (
	"context"
	"encoding/json"
	"fmt"
	"log"
	"os"
	"strings"

	"github.com/urfave/cli/v3"
)

var (
	isDebug  bool   = false // 是否输出调试信息
	isJSON   bool   = false // 是否以json格式输出
	isIgnore bool   = false // 是否忽略大搜索
	keyword  string         // 搜索关键字
)

// InitJSONFlag 初始化json格式输出标志(可复用所以用函数来初始化): --json
func InitJSONFlag() *cli.BoolFlag {
	return &cli.BoolFlag{
		Name:        "json",
		Usage:       "print result in json format",
		Value:       isJSON,
		Destination: &isJSON,
	}
}

// PrintDebuInfo 输出调试信息
func PrintDebuInfo(printFn func()) {
	if isDebug {
		printFn()
	}
}

func PrintAsJSON(original string, formatted string) {
	// {"original":"Hello World", "formatted": "HELLO WORLD", isDebug:true}
	data := make(map[string]any)
	data["original"] = original
	data["formatted"] = strings.ToUpper(original)
	data["isDebug"] = isDebug
	json, err := json.Marshal(data)
	if err != nil {
		fmt.Println("[Error]json encodeing:", err)
		return
	}
	fmt.Println(string(json))
}

func InitUpperCommand() *cli.Command {
	return &cli.Command{
		Name:    "upper",
		Aliases: []string{"u", "up"},
		Usage:   "format text to upper case",
		Flags: []cli.Flag{
			InitJSONFlag(),
		},
		Action: func(ctx context.Context, cmd *cli.Command) error {
			// textfmt contains -k hello Mock Long String Hello
			// 如果不加双引号, Mock Long String Hello 是多个参数
			// cmd.Args().First() 或 cmd.Args().First() 只会获取第一个参数 Mock
			// 所以应该获取所有参数并拼接字符串
			original := strings.Join(cmd.Args().Slice(), " ")
			formatted := strings.ToUpper(original)

			PrintDebuInfo(func() {
				fmt.Println("[original]", original)
				fmt.Println("[formatted]", formatted)
			})

			if isJSON {
				PrintAsJSON(original, formatted)
			} else {
				fmt.Println(formatted)
			}

			return nil
		},
	}
}

func InitLowerCommand() *cli.Command {
	return &cli.Command{
		Name:    "lower",
		Aliases: []string{"l", "lo"},
		Usage:   "format text to lower case",
		Flags: []cli.Flag{
			InitJSONFlag(),
		},
		Action: func(ctx context.Context, cmd *cli.Command) error {
			original := strings.Join(cmd.Args().Slice(), " ")
			formatted := strings.ToLower(original)

			PrintDebuInfo(func() {
				fmt.Println("[original]", original)
				fmt.Println("[formatted]", formatted)
			})

			if isJSON {
				PrintAsJSON(original, formatted)
			} else {
				fmt.Println(formatted)
			}
			return nil
		},
	}
}

func InitContainsCommand() *cli.Command {
	return &cli.Command{
		Name:    "contains",
		Aliases: []string{"c"},
		Usage:   "search string in text",
		Flags: []cli.Flag{
			// --ignore/-i: 忽略大小写
			&cli.BoolFlag{
				Name:        "ignore",
				Aliases:     []string{"i"},
				Usage:       "ignore case when search",
				Value:       isIgnore,
				Destination: &isIgnore,
			},

			// --keyword/-k: 要搜索的关键字
			&cli.StringFlag{
				Name:        "keyword",
				Aliases:     []string{"k"},
				Usage:       "ignore case when search",
				Value:       "",
				Destination: &keyword,
			},
		},
		Action: func(ctx context.Context, cmd *cli.Command) error {
			original := strings.Join(cmd.Args().Slice(), " ")

			// 验证 --keyword/-k 选项
			if len(keyword) == 0 {
				return fmt.Errorf("[Error]keyword is empty")
			}

			PrintDebuInfo(func() {
				fmt.Println("[original]", original)
				fmt.Println("[keyword]", keyword)
			})

			// 是否忽略大小写选项
			if isIgnore {
				isContainsIgnoreCase := strings.Contains(strings.ToLower(original), strings.ToLower(keyword))
				fmt.Println(isContainsIgnoreCase) // "true" or "false"
			} else {
				isContains := strings.Contains(original, keyword)
				fmt.Println(isContains) // "true" or "false"
			}

			return nil
		},
	}
}

func InitCommand() *cli.Command {
	cmd := &cli.Command{
		Name:  "textfmt",
		Usage: "format text",
		Flags: []cli.Flag{
			// 全局选项: --debug/-d
			&cli.BoolFlag{
				Name:        "debug",
				Aliases:     []string{"d"},
				Usage:       "print debug information",
				Value:       isDebug,
				Destination: &isDebug,
			},
		},
		// 初始化子命令
		Commands: []*cli.Command{
			InitUpperCommand(),
			InitLowerCommand(),
			InitContainsCommand(),
		},
		// Action: 根命令没有 Action, 默认执行: cmd --help
	}
	return cmd
}

func main() {
	cmd := InitCommand()
	if err := cmd.Run(context.Background(), os.Args); err != nil {
		log.Fatal(err)
	}
}
```

```go [cmd/root.go]
package cmd

import (
	"encoding/json"
	"fmt"
	"os"
	"strings"

	"github.com/spf13/cobra"
)

// 全局选项 & 可复用选项
var (
	isDebug bool = false // --debug/-d 全局选项
	isJSON  bool = false // --json 局部选项
)

// 根命令
var rootCmd = &cobra.Command{
	Use:   "textfmt2",
	Short: "a cli tool for formatting strings",
	Long: `a cli tool for formatting strings:
	upper: format string to upper case
	lower: format string to lower case
	contains: check if string contains a substring`,
	// 根命令没有设置 Run/RunE 默认执行 cmd --help
}

// init 模块初始化钩子函数
func init() {
	// 初始化 --debug 选项
	InitDebugFlag(upperCmd)
	InitDebugFlag(lowerCmd)
	InitDebugFlag(containsCmd)

	// 初始化 --json 选项
	InitIsJSONFlag(upperCmd)
	InitIsJSONFlag(lowerCmd)
}

// Execute 执行命令
func Execute() {
	err := rootCmd.Execute()
	if err != nil {
		os.Exit(1)
	}
}

// InitDebugFlag 初始化 --debug 选项(复用)
// 与 urfave/cli 不同的是: 使用子命令时, 默认不会继承全局选项
// 也就是说: 给 rootCmd 设置 debugFlag, 使用 rootCmd upper 时无法直接使用 --debug 选项
func InitDebugFlag(cmd *cobra.Command) {
	cmd.Flags().BoolVarP(&isDebug, "debug", "d", isDebug, "Print debug infomation")
}

// InitIsJSONFlag 初始化 --json 选项(复用)
func InitIsJSONFlag(cmd *cobra.Command) {
	cmd.Flags().BoolVar(&isJSON, "json", isJSON, "Print result as json format")
}

// PrintDebuInfo 输出调试信息(复用)
func PrintDebuInfo(printFn func()) {
	if isDebug {
		printFn()
	}
}

// PrintAsJSON 输出json 格式的信息
func PrintAsJSON(original string, formatted string) {
	// {"original":"Hello World", "formatted": "HELLO WORLD", isDebug:true}
	data := make(map[string]any)
	data["original"] = original
	data["formatted"] = strings.ToUpper(original)
	data["isDebug"] = isDebug
	json, err := json.Marshal(data)
	if err != nil {
		fmt.Println("[Error]json encodeing:", err)
		return
	}
	fmt.Println(string(json))
}
```

```go [cmd/contains.go]
package cmd

import (
	"fmt"
	"strings"

	"github.com/spf13/cobra"
)

var (
	isIgnore bool   = false // 是否忽略大搜索
	keyword  string         // 搜索关键字
)

// containsCmd represents the contains command
var containsCmd = &cobra.Command{
	Use:     "contains",
	Aliases: []string{"c"},
	Short:   "check if string contains a keyword",
	Long: `check if string contains a keyword|
	Example: textfm2 contains "Hello World" -k="world"
	Output: false

	Example: textfm2 contains "Hello World" -i -k="world"
	Output: true`,
	RunE: func(cmd *cobra.Command, args []string) error {
		original := strings.Join(args, " ")

		// 验证 --keyword/-k 选项
		if len(keyword) == 0 {
			return fmt.Errorf("[Error]keyword is empty")
		}

		PrintDebuInfo(func() {
			fmt.Println("[original]", original)
			fmt.Println("[keyword]", keyword)
		})

		// 是否忽略大小写选项
		if isIgnore {
			isContainsIgnoreCase := strings.Contains(strings.ToLower(original), strings.ToLower(keyword))
			fmt.Println(isContainsIgnoreCase) // "true" or "false"
		} else {
			isContains := strings.Contains(original, keyword)
			fmt.Println(isContains) // "true" or "false"
		}

		return nil
	},
}

func init() {
	containsCmd.Flags().BoolVarP(&isIgnore, "ignore", "i", isIgnore, "Ignore case when search")
	containsCmd.Flags().StringVarP(&keyword, "keyword", "k", keyword, "Keyword to search")
	rootCmd.AddCommand(containsCmd)
}
```
